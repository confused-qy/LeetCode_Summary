和动态规划无关，内存为O(1)的巧妙做法
``` cpp
class Solution {
public:
    int longestValidParentheses(string s) {
        // 注意，是最长连续！
        // 计算左括号和右括号的数量
        int left = 0;
        int right = 0;
        int currMax = 0;

        for (int i = 0; i < s.size(); i++){
            if(s[i] == '(') {
                left += 1;
            }
            else {
                right += 1;
            }

            // 如果左右括号数量相等，现在是合法的字串，看看长度如何
            if (left == right) {
                currMax = max(currMax, left + right);
            }
            // 如果左括号数量小于有括号，该字串一定不可能合法，舍弃重新计算左右括号
            if (left < right) {
                left = 0;
                right = 0;
            }
        }

        // 上述做法会遗漏类似“（（）”这种，左括号总是比右括号多的情况。要从右往左再来一遍
        left = right = 0;
        for (int i = s.size() - 1; i >= 0; i--){
            if(s[i] == '(') {
                left += 1;
            }
            else {
                right += 1;
            }

            if (left == right) {
                currMax = max(currMax, left + right);
            }
            
            if (right < left) {
                left = 0;
                right = 0;
            }
        }

        return currMax;
    }
};
```

动态规划：
``` cpp
class Solution {
public:
    int longestValidParentheses(string s) {
        // dp[i]：以第i位为结尾的话，可以达到的最长的子字符串长度
        vector<int> dp(s.size(), 0);
        int currMax = 0;

        for (int i = 1; i < s.size(); i++){
            // s[i] 是左括号，则不可能作为子字符串结尾
            
            // s[i]是右括号
            if (s[i] == ')'){
                // s[i-1]是左括号，那么就是在dp[i-2]基础上加2
                if (i-2 >= 0 && s[i-1] == '('){
                    dp[i] = dp[i-2] + 2;
                }
                // s[i-1]是右括号，i-1所在的最长子字符串前面是不是(
                // 如果是，还要加上再前面的长度
                else {
                    if (i-dp[i-1]-1 >= 0 && s[i-dp[i-1]-1] == '('){
                        dp[i] = dp[i-1] + 2;
                        if (i-dp[i-1]-2 >= 0){
                            dp[i] += dp[i-dp[i-1]-2];
                        }
                    }
                }

                currMax = max(dp[i], currMax);
            }
        }

        return currMax;
    }
};
```
