和动态规划无关，内存为O(1)的巧妙做法
``` cpp
class Solution {
public:
    int longestValidParentheses(string s) {
        // 注意，是最长连续！
        // 计算左括号和右括号的数量
        int left = 0;
        int right = 0;
        int currMax = 0;

        for (int i = 0; i < s.size(); i++){
            if(s[i] == '(') {
                left += 1;
            }
            else {
                right += 1;
            }

            // 如果左右括号数量相等，现在是合法的字串，看看长度如何
            if (left == right) {
                currMax = max(currMax, left + right);
            }
            // 如果左括号数量小于有括号，该字串一定不可能合法，舍弃重新计算左右括号
            if (left < right) {
                left = 0;
                right = 0;
            }
        }

        // 上述做法会遗漏类似“（（）”这种，左括号总是比右括号多的情况。要从右往左再来一遍
        left = right = 0;
        for (int i = s.size() - 1; i >= 0; i--){
            if(s[i] == '(') {
                left += 1;
            }
            else {
                right += 1;
            }

            if (left == right) {
                currMax = max(currMax, left + right);
            }
            
            if (right < left) {
                left = 0;
                right = 0;
            }
        }

        return currMax;
    }
};
```
