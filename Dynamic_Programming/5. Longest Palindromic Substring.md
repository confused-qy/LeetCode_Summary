中心扩散方法
``` cpp
class Solution {
public:
    string longestPalindrome(string s) {
        // 中心扩散方法，不算是动态规划

        // 最长回文子串的开始与结束
        int begin = 0;
        int maxLen = 1;
        for (int i = 0; i < s.size() - 1; i++) {
            int len1 = expandAroundCenter(s, i, i);
            int len2 = expandAroundCenter(s, i, i + 1);
            int cur = max(len1, len2);
            if (cur > maxLen) {
                maxLen = cur;
                begin = i - (cur - 1) / 2;
            }
        }

        return s.substr(begin, maxLen);
    }

    // 在某一位最长的回文子串
    int expandAroundCenter(string s, int left, int right) {
        while (left >= 0 && right  <= s.size() - 1 &&
               s[left] == s[right]) {
            left -= 1;
            right += 1;
        }
        // 循环多走了一格，要减掉
        return right - left - 1;
    }
};
```

``` cpp
class Solution {
public:
    string longestPalindrome(string s) {
        // Manacher算法
        // 先在字符串里插入#，确保都是奇数对称
        string t = "#";
        for (char c : s) {
            t += c;
            t += "#";
        }

        // 通过中心扩散法能够扩散的最右边的下标
        int maxRight = 0;
        // 对应的中心的下标
        int center = 0;

        int maxLen = 1;
        int begin = 0;
        vector<int> p(t.size(), 0);

        for (int i = 0; i < t.size(); i++) {
            if (i < maxRight) {
                int mirror = 2 * center - i;
                // 状态转移方程
                p[i] = min(p[mirror], maxRight - i);
            }
            // 用中心扩散来更新p[i]数值
            while (i - p[i] - 1 >= 0 && i + p[i] + 1 < t.size() &&
                   t[i - p[i] - 1] == t[i + p[i] + 1]) {

                p[i]++;
            }

            // 更新中心和边界
            if (i + p[i] > maxRight) {
                maxRight = i + p[i];
                center = i;
            }

            // 更新最长回文子串
            if (p[i] > maxLen) {
                maxLen = p[i];
                begin = (i - maxLen) / 2; // 要没有插入#时的原本坐标！
            }
        }

        return s.substr(begin, maxLen);
    }
};
```
