class Solution {
public:
    int minDistance(string word1, string word2) {
        int n = word1.size();
        int m = word2.size();

        // 有一个是空数组的情况
        if (n * m == 0) {
            return n + m;
        }

        // dp[i][j]: word1到第i位和word2到第j位的最小编辑距离
        vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));

        // 初始化第一行和第一列
        for (int i = 0; i < n + 1; i++) {
            dp[i][0] = i;
        }
        for (int j = 0; j < m + 1; j++) {
            dp[0][j] = j;
        }

        for (int i = 1; i < n + 1; i++) {
            for (int j = 1; j < m + 1; j++) {
                // 如果这一位相同
                if (word1[i - 1] == word2[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1];
                }
                // 如果这一位不同
                else {
                    // 注意min()不能一次传进去三个参数！必须min({})
                    dp[i][j] =
                        1 + min({dp[i - 1][j - 1], dp[i - 1][j], dp[i][j - 1]});
                }
            }
        }
        return dp[n][m];
    }
};
