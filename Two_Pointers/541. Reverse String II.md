class Solution {
public:
    string reverseStr(string s, int k) {
        // 模拟，反转每隔下标从2k开始，长度为k的子串
        for (int i = 0; i < s.size(); i += 2 * k) {
            // 注意：s.size()返回值不是int而是size_t，是无符号的
            reverse(s.begin() + i, s.begin() + min(i + k, (int)s.size()));
        }
        return s;
    }
};
