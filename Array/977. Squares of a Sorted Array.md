``` cpp
class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {
        // 首先，找到最后一个负数的位置
        int neg = nums.size() - 1;
        for (int i = 0; i < nums.size(); i++) {
            if (nums[i] >= 0) {
                neg = i - 1;
                break;
            }
        }

        // 这个数左边平方后是降序，右边平方后是升序
        int i = neg;
        int j = neg + 1;
        vector<int> result;
        while (i >= 0 || j < nums.size()) {
            if (i == -1) {
                result.push_back(nums[j] * nums[j]);
                j++;
            }
            else if (j == nums.size()) {
                result.push_back(nums[i] * nums[i]);
                i--;
            }
            else if (-nums[i] <= nums[j]) {
                result.push_back(nums[i] * nums[i]);
                i--;
            }
            else {
                result.push_back(nums[j] * nums[j]);
                j++;
            }
        }

        return result;
    }
};
```
