class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        // 双指针从左和右出发，如果左边是val就把右边的复制过来，右边左移
        // 如果复制过来的还是val就继续复制
        int left = 0;
        int right = nums.size() - 1;

        while (left <= right) {
            if (nums[left] == val) {
                nums[left] = nums[right];
                right --;
            }
            else {
                left ++;
            }
        }

        return left;
    }
};
