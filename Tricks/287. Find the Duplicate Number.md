class Solution {
public:
    int findDuplicate(vector<int>& nums) {
        // Floyd判圈算法
        
        // 从0开始，每一步走向的下一个下标是nums[i]
        // 由于每个节点都有一条出边，且节点有限，不断沿边走必然形成环
        // 重复数字作为下标的节点位置至少有两个入边，因此成为环的开始
        // 快慢指针会在环里面相遇
        int slow = 0; // 不可能有箭头指向0，所以从0开始是安全的
        int fast = 0;
        while (1) {
            slow = nums[slow];
            fast = nums[fast];
            fast = nums[fast];
            if (slow == fast) {
                break;
            }
        }
        // 相遇下标为slow的位置

        // 设从0到环开始步数为a，环长为L
        // 环开始到相遇点步数为b，相遇点回到环开始步数为c
        // b + c = L
        // 慢指针步数为a + b
        // 快指针步数为2(a + b) = a + b + kL
        // 整理得a = (k - 1)L + c
        // 两个同速度指针，一个从0开始，一个从快慢指针的相遇点开始，就会在环开始处相遇
        // 此时的位置就是答案！
         int a1 = 0;
         int a2 = slow;
         while (a1 != a2) {
            a1 = nums[a1];
            a2 = nums[a2];
         }

         return a1;
        
    }
};
